cmake_minimum_required(VERSION 3.16)
project(potasms VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    # Preserve UTF-8 fixtures and match the Flutter plugin's float contract.
    add_compile_options(/utf-8 /fp:strict)
endif()
file(GLOB PCRE2_SOURCES CONFIGURE_DEPENDS Sources/CPCRE2/*.c)
add_library(potasms SHARED ${PCRE2_SOURCES} Sources/CPOTASMS/rules.cpp Sources/CPOTASMS/potasms.cpp Sources/CUtf8Proc/utf8proc.c)
target_include_directories(potasms PUBLIC Sources/CPOTASMS/include PRIVATE Sources/CUtf8Proc Sources/CPCRE2)
target_compile_definitions(potasms PRIVATE UTF8PROC_STATIC)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(potasms PRIVATE -ffp-contract=off -Wall -Wextra)
endif()
set_target_properties(potasms PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
include(CTest)
if(BUILD_TESTING)
    add_executable(potasms_native_tests Tests/Native/core_tests.cpp)
    target_link_libraries(potasms_native_tests PRIVATE potasms)
    add_test(NAME native_core COMMAND potasms_native_tests)
endif()
add_executable(potasms_native_bench Tests/Native/benchmark.cpp)
target_link_libraries(potasms_native_bench PRIVATE potasms)
